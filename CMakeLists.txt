cmake_minimum_required(VERSION 3.24)
project(Quanta VERSION 0.1.0 LANGUAGES CXX)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
  message(FATAL_ERROR "Use an out-of-source build: cmake -S . -B build/debug")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(QUANTA_BUILD_TESTS "Build automated tests and fixture tool" ON)
option(QUANTA_BUILD_BENCHMARKS "Build storage benchmarks and fixture tool" OFF)
option(QUANTA_ENABLE_ASAN "Instrument Quanta with AddressSanitizer" OFF)
option(QUANTA_ENABLE_UBSAN "Instrument Quanta with UndefinedBehaviorSanitizer" OFF)
include(cmake/Dependencies.cmake)
include(cmake/QuantaOptions.cmake)
add_library(quanta_storage src/execution/vector_batch.cpp src/storage/parquet_scanner.cpp)
add_library(Quanta::storage ALIAS quanta_storage)
target_include_directories(quanta_storage PUBLIC include)
target_link_libraries(quanta_storage PUBLIC Arrow::arrow_shared Parquet::parquet_shared)
quanta_target_options(quanta_storage)

if(QUANTA_BUILD_TESTS OR QUANTA_BUILD_BENCHMARKS)
  add_library(quanta_fixtures STATIC tests/fixtures/parquet_fixture.cpp)
  target_include_directories(quanta_fixtures PUBLIC tests/fixtures)
  target_link_libraries(quanta_fixtures PUBLIC Quanta::storage)
  quanta_target_options(quanta_fixtures)
endif()
configure_file(cmake/build_info.h.in generated/quanta/build_info.h @ONLY)
add_library(quanta_cli STATIC app/cli.cpp)
target_include_directories(quanta_cli PUBLIC app PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(quanta_cli PRIVATE QUANTA_BUILD_TYPE="$<CONFIG>")
target_link_libraries(quanta_cli PUBLIC Quanta::storage)
quanta_target_options(quanta_cli)
add_executable(quanta app/main.cpp)
target_link_libraries(quanta PRIVATE quanta_cli)
quanta_target_options(quanta)
if(QUANTA_BUILD_TESTS OR QUANTA_BUILD_BENCHMARKS)
  add_executable(quanta_make_fixture tools/make_fixture.cpp)
  target_link_libraries(quanta_make_fixture PRIVATE quanta_fixtures)
  quanta_target_options(quanta_make_fixture)
endif()
if(QUANTA_BUILD_TESTS)
  enable_testing()
  add_executable(quanta_tests
    tests/unit/vector_batch_test.cpp tests/unit/parse_integer_test.cpp
    tests/integration/parquet_scanner_test.cpp tests/integration/cli_test.cpp)
  target_link_libraries(quanta_tests PRIVATE quanta_fixtures quanta_cli GTest::gtest_main)
  quanta_target_options(quanta_tests)
  include(GoogleTest)
  gtest_discover_tests(quanta_tests DISCOVERY_MODE PRE_TEST)
  add_test(NAME CliProcess COMMAND "${CMAKE_COMMAND}"
    "-DQUANTA=$<TARGET_FILE:quanta>" "-DFIXTURE_TOOL=$<TARGET_FILE:quanta_make_fixture>"
    "-DTEST_DIR=${CMAKE_CURRENT_BINARY_DIR}" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/cli_process_test.cmake")
endif()
if(QUANTA_BUILD_BENCHMARKS)
  add_executable(quanta_scan_benchmark benchmarks/micro/parquet_scan_benchmark.cpp)
  target_link_libraries(quanta_scan_benchmark PRIVATE quanta_fixtures benchmark::benchmark)
  quanta_target_options(quanta_scan_benchmark)
endif()
